import { SQSClient, GetQueueUrlCommand, ReceiveMessageCommand, DeleteMessageCommand } from "@aws-sdk/client-sqs";
import S3Service from '../services/s3.service.js';
import FigmaApiClient from '../clients/figma-api.client.js';
import DesireAiApiClient from '../clients/desire-ai-api.client.js';
import logger from "../config/logger.js";
import * as dotenv from 'dotenv';

dotenv.config();

// --- Configuration (remains the same) ---
const queueName = process.env.IMAGE_GEN_QUEUE!;
const sqsClient = new SQSClient({ /* ... */ });
const s3Service = new S3Service();

// =================================================================
// CHANGE #1: The "Work" is separated into its own function.
// This makes the code readable, maintainable, and easy to test.
// =================================================================
async function handleMessage(messageBody: string) {
    logger.info(`[Handler] Starting to process message...`);
    
    // 1. Parse the incoming message.
    const job = JSON.parse(messageBody);
    const { bucket, key } = job;
    if (!bucket || !key) {
        throw new Error("Message is invalid. It must contain a 'bucket' and 'key'.");
    }

    // 2. Fetch the data from S3.
    const objects = await s3Service.getObjectsByPrefix(bucket, key);
    if (objects.length === 0) {
        throw new Error(`Could not find object with key '${key}' in bucket '${bucket}'.`);
    }
    logger.info(`[Handler] Successfully retrieved '${objects[0].key}' from S3.`);

    // 3. Send the data to the Figma API.
    const figmaResult = await FigmaApiClient.processImage(objects[0]);

    // 4. Send the final result back to the main Desire AI backend.
    await DesireAiApiClient.updateCampaignAsset({ originalJob: job, figmaResult });

    logger.info(`[Handler] Successfully completed processing for message.`);
}

// =================================================================
// CHANGE #2 & #3: The polling logic is consolidated and made resilient.
// =================================================================
async function startListener() {
    logger.info(`Listener starting... Attempting to find queue: ${queueName}`);
    const { QueueUrl } = await sqsClient.send(new GetQueueUrlCommand({ QueueName: queueName }));
    if (!QueueUrl) {
        throw new Error(`Could not find SQS queue named '${queueName}'.`);
    }
    logger.info(`Successfully connected to queue. Now polling for messages...`);

    while (true) {
        try {
            const { Messages } = await sqsClient.send(new ReceiveMessageCommand({ /* ... */ }));

            if (Messages && Messages.length > 0) {
                const message = Messages[0];
                if (message.Body) {
                    // This inner try/catch is the key to resiliency.
                    // It isolates errors to a single message.
                    try {
                        // Delegate the actual work to our clean, testable function.
                        await handleMessage(message.Body);

                        // If the work succeeds, we delete the message.
                        await sqsClient.send(new DeleteMessageCommand({
                            QueueUrl,
                            ReceiptHandle: message.ReceiptHandle!,
                        }));
                        logger.info(`[Poller] Message processed and deleted from queue.`);

                    } catch (processingError: any) {
                        // If handleMessage fails, we log it but DO NOT delete the message.
                        // SQS will automatically make it available for a retry later.
                        logger.error(`[Poller] Failed to process message. It will be retried. Error: ${processingError.message}`);
                    }
                }
            }
        } catch (pollingError: any) {
            logger.error(`[Poller] A critical error occurred while polling SQS: ${pollingError.message}`);
            await new Promise(resolve => setTimeout(resolve, 5000)); // Wait before trying again
        }
    }
}

startListener().catch(err => {
    logger.error("Listener failed to start:", err);
    process.exit(1);
});
